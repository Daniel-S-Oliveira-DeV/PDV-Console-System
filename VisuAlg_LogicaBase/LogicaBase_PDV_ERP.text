procedimento CadastrarProduto()  

inicio  

   Cabecalho()  

   escreval("--- CADASTRAR NOVO PRODUTO ---")  

 

   // NOVA LINHA: Chama o catálogo antes de iniciar o cadastro  

   ListarProdutos()  

   escreval("")  

 

   se qtdProd >= 50 entao  

      escreval("ERRO: Limite máximo de 50 produtos atingido.")  

   senao  

      qtdProd <- qtdProd + 1  

      escreva("Código numérico (Novo): ")  

      leia(vetProdCod[qtdProd])  

      escreva("Nome/Descrição do Produto: ")  

      leia(vetProdNome[qtdProd])  

      escreva("Custo Unitário de Fábrica R$: ")  

      leia(vetProdCusto[qtdProd])  

      escreva("Preço Unitário de Venda R$: ")  

      leia(vetProdPreco[qtdProd])  

      escreva("Quantidade Inicial em Estoque: ")  

      leia(vetProdEstoque[qtdProd])  

      escreva("Alerta de Estoque Mínimo: ")  

      leia(vetProdEstqMin[qtdProd])  

 

      // Adiciona o custo desta aquisição inicial às despesas  

      despesasCompras <- despesasCompras + (vetProdCusto[qtdProd] * vetProdEstoque[qtdProd])  

 

      escreval("")  

      escreval("Produto cadastrado com sucesso!")  

   fimse  

   Pausar()  

fimprocedimento  

 

// NOVA FUNCIONALIDADE: APENAS AUMENTAR ESTOQUE EXISTENTE  

procedimento EntradaMercadoria()  

inicio  

   Cabecalho()  

   escreval("--- REPOSIÇÃO DE ESTOQUE (COMPRAS DO FORNECEDOR) ---")  

   ListarProdutos()  

   escreva("Digite o CÓDIGO do produto que chegou do fornecedor: ")  

   leia(codBusca)  

   idxProd <- 0  

 

   para i de 1 ate qtdProd faca  

      se vetProdCod[i] = codBusca entao  

         idxProd <- i  

      fimse  

   fimpara  

 

   se idxProd <> 0 entao  

      escreval("Produto localizado: ", vetProdNome[idxProd])  

      escreva("Quantidade recebida para somar ao estoque: ")  

      leia(qtdDesejada)  

 

      vetProdEstoque[idxProd] <- vetProdEstoque[idxProd] + qtdDesejada  

      despesasCompras <- despesasCompras + (vetProdCusto[idxProd] * qtdDesejada)  

 

      escreval("Estoque atualizado para: ", vetProdEstoque[idxProd], " unidades.")  

   senao  

      escreval("ERRO: Produto não encontrado no sistema.")  

   fimse  

   Pausar()  

fimprocedimento  

 

procedimento AbrirFecharCaixa()  

inicio  

   Cabecalho()

      escreval("--- ABERTURA DE CAIXA ---")  

      escreva("Informe o valor do fundo de caixa (Troco Inicial) R$: ")  

      leia(fundoCaixa)  

      caixaAberto <- verdadeiro  

      escreval("Caixa aberto com sucesso! Boas vendas.")  

   senao  

      escreval("--- FECHAMENTO DE CAIXA (Z-READING) ---")  

      escreval(" Fundo Inicial (Troco) : R$ ", fundoCaixa:4:2)  

      escreval(" Entradas em Dinheiro  : R$ ", totalDinheiro:4:2)  

      escreval("-----------------------------------------")  

      escreval(" TOTAL ESPERADO GAVETA : R$ ", (fundoCaixa + totalDinheiro):4:2)  

      escreval(" ")  

      escreval(" Total Cartão          : R$ ", totalCartao:4:2)  

      escreval(" Total PIX             : R$ ", totalPix:4:2)  

      escreval("-----------------------------------------")  

      caixaAberto <- falso  

      totalDinheiro <- 0; totalCartao <- 0; totalPix <- 0  

      escreval("Caixa fechado com sucesso!")  

   fimse  

   Pausar()  

fimprocedimento

procedimento PDV_FrenteDeCaixa()  

inicio  

   se caixaAberto = falso entao  

      Cabecalho()  

      escreval("ERRO: O Caixa está fechado! Solicite a abertura ao ADMIN.")  

      Pausar()  

   senao  

      Cabecalho()  

      escreval("--- NOVA VENDA ---")  

      idVendaAtual <- qtdVendas + 1  

      subtotal <- 0  

      idxCli <- 0  

 

      // CRM: Identificação do Cliente  

      escreva("Informe o CPF do cliente (ou deixe vazio para anônimo): ")  

      leia(cpfBusca)  

 

      se cpfBusca <> "" entao  

         para i de 1 ate qtdCli faca  

            se vetCliCPF[i] = cpfBusca entao  

               idxCli <- i  

            fimse  

         fimpara  

 

         se idxCli = 0 entao  

            qtdCli <- qtdCli + 1  

            idxCli <- qtdCli  

            vetCliCPF[idxCli] <- cpfBusca  

            escreva("Cliente não cadastrado. Informe o Nome: ")  

            leia(vetCliNome[idxCli])  

            vetCliGasto[idxCli] <- 0  

            escreval("Cliente cadastrado com sucesso!")  

         senao  

            escreval("Bem-vindo de volta, ", vetCliNome[idxCli], "!")  

         fimse  

      fimse  

 

      // Laço de Itens  

      continuar <- "S"  

      enquanto (continuar = "S") ou (continuar = "s") faca  

         Cabecalho()  

         ListarProdutos()  

         escreval("=== CARRINHO ATUAL: R$ ", subtotal:4:2, " ===")  

         escreva("Digite o CÓDIGO do Produto: ")  

         leia(codBusca)  

 

         idxProd <- 0  

         para i de 1 ate qtdProd faca  

            se vetProdCod[i] = codBusca entao  

               idxProd <- i  

            fimse  

         fimpara  

 

         se idxProd <> 0 entao  

            escreval("Produto: ", vetProdNome[idxProd], " | R$ ", vetProdPreco[idxProd]:4:2)  

            escreva("Quantidade: ")  

            leia(qtdDesejada)  

 

            se qtdDesejada <= vetProdEstoque[idxProd] entao  

               vetProdEstoque[idxProd] <- vetProdEstoque[idxProd] - qtdDesejada  

               subtotal <- subtotal + (vetProdPreco[idxProd] * qtdDesejada)  

 

               qtdItens <- qtdItens + 1  

               vetItemVenID[qtdItens] <- idVendaAtual  

               vetItemNome[qtdItens] <- vetProdNome[idxProd]  

               vetItemQtd[qtdItens] <- qtdDesejada  

               vetItemCustoUni[qtdItens] <- vetProdCusto[idxProd]  

               vetItemPrecoUni[qtdItens] <- vetProdPreco[idxProd]  

 

               escreval(">> Item adicionado!")  

            senao  

               escreval(">> ERRO: Estoque insuficiente! (Disponível: ", vetProdEstoque[idxProd], ")")  

            fimse  

         senao  

            escreval(">> ERRO: Código Inválido!")  

         fimse  

 

         escreva("Passar outro item? (S/N): ")  

         leia(continuar)  

      fimenquanto  

 

      // Finalização e Pagamento  

      se subtotal > 0 entao  

         Cabecalho()  

         escreval("--- FINALIZAÇÃO DA VENDA ---")  

         escreval("Subtotal: R$ ", subtotal:4:2)  

         escreva("Desconto R$ (0 para nenhum): ")  

         leia(desconto)  

         totalPagar <- subtotal - desconto  

         se totalPagar < 0 entao  

            totalPagar <- 0  

         fimse  

 

         escreval("TOTAL A PAGAR: R$ ", totalPagar:4:2)  

         escreval("Forma de Pagamento: [1] Dinheiro  [2] Cartão  [3] PIX")  

         leia(formaPag)  

 

         se formaPag = "1" entao  

            vetVenPagamento[idVendaAtual] <- "DINHEIRO"  

            escreva("Valor recebido do cliente R$: ")  

            leia(valorRecebido)  

            enquanto valorRecebido < totalPagar faca  

               escreval("Valor insuficiente!")  

               escreva("Valor recebido do cliente R$: ")  

               leia(valorRecebido)  

            fimenquanto  

            troco <- valorRecebido - totalPagar  

            totalDinheiro <- totalDinheiro + totalPagar  

            escreval("TROCO A DEVOLVER: R$ ", troco:4:2)  

         senao  

            se formaPag = "2" entao  

               vetVenPagamento[idVendaAtual] <- "CARTAO"  

               totalCartao <- totalCartao + totalPagar  

               escreval("Transação aprovada na maquininha!")  

            senao  

               vetVenPagamento[idVendaAtual] <- "PIX"  

               totalPix <- totalPix + totalPagar  

               escreval("PIX recebido com sucesso!")  

            fimse  

         fimse  

 

         // Salva a Venda  

         qtdVendas <- idVendaAtual  

         vetVenID[idVendaAtual] <- idVendaAtual  

         vetVenTotal[idVendaAtual] <- totalPagar  

 

         // Atualiza CRM  

         se idxCli <> 0 entao  

            vetCliGasto[idxCli] <- vetCliGasto[idxCli] + totalPagar  

         fimse  

 

         escreval("===========================================")  

         escreval(" CUPOM FISCAL GERADO COM SUCESSO! ")  

         escreval("===========================================")  

      senao  

         escreval("Venda cancelada. Nenhum item registrado.")  

      fimse  

      Pausar()  

   fimse  

fimprocedimento

procedimento RelatoriosAdmin()  

inicio  

   Cabecalho()  

   escreval("--- PAINEL DE CONTROLE FINANCEIRO E CRM ---")  

 

   // 1. Alerta de Stock Crítico  

   escreval(">> ALERTAS DE ESTOQUE:")  

   para i de 1 ate qtdProd faca  

      se vetProdEstoque[i] <= vetProdEstqMin[i] entao  

         escreval("   [CRÍTICO] ", vetProdNome[i], " - Restam apenas ", vetProdEstoque[i], " un.")  

      fimse  

   fimpara  

   escreval("")  

 

   // 2. Cálculo de Lucro Líquido (Bottom Line)  

   faturamentoTotal <- 0  

   custoTotalVendido <- 0  

 

   para i de 1 ate qtdVendas faca  

      faturamentoTotal <- faturamentoTotal + vetVenTotal[i]  

   fimpara  

 

   para i de 1 ate qtdItens faca  

      custoTotalVendido <- custoTotalVendido + (vetItemCustoUni[i] * vetItemQtd[i])  

   fimpara  

 

   lucroTotal <- faturamentoTotal - custoTotalVendido - despesasCompras  

 

   escreval(">> BALANÇO FINANCEIRO DA LOJA:")  

   escreval("   Faturamento Bruto (Vendas) : R$ ", faturamentoTotal:4:2)  

   escreval("   Custo Mercadorias Vendidas : R$ ", custoTotalVendido:4:2)  

   escreval("   Despesas c/ Novas Compras  : R$ ", despesasCompras:4:2)  

   escreval("   ---------------------------------------")  

   escreval("   LUCRO LÍQUIDO REAL         : R$ ", lucroTotal:4:2)  

   escreval("")  

 

   // 3. CRM - Melhores Clientes  

   escreval(">> BASE DE CLIENTES CADASTRADOS:")  

   se qtdCli = 0 entao  

      escreval("   Nenhum cliente cadastrado ainda.")  

   senao  

      para i de 1 ate qtdCli faca  

         escreval("   Cliente: ", vetCliNome[i], " | Total Gasto: R$ ", vetCliGasto[i]:4:2)  

      fimpara  

   fimse  

   Pausar()  

fimprocedimento
